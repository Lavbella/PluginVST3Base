cmake_minimum_required(VERSION 3.15)

project(IbanezTubeScreamer VERSION 1.0.0 LANGUAGES C CXX) 

# ==========================================
# GLOBAL JUCE DEFINITIONS (Before add_subdirectory)
# ==========================================
set(JUCE_WEB_BROWSER 0 CACHE INTERNAL "")
set(JUCE_USE_CURL 0 CACHE INTERNAL "")
set(JUCE_VST3_CAN_REPLACE_VST2 0 CACHE INTERNAL "")

# Note: To guarantee reproducibility (reproducible CI), the ideal approach is to pin
# the JUCE version as a Git submodule pointing to a specific commit or tag.
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../JUCE" "${CMAKE_CURRENT_BINARY_DIR}/JUCE")

# ==========================================
# AUDIO COMPONENT CONFIGURATION
# ==========================================
juce_add_plugin(IbanezTubeScreamer
    COMPANY_NAME "Lavbella"
    PLUGIN_MANUFACTURER_CODE MPlg
    PLUGIN_CODE TScr
    FORMATS CLAP VST3                   
    PRODUCT_NAME "Paudiog01"
    IS_SYNTH FALSE                      
)

# Source files
target_sources(IbanezTubeScreamer PRIVATE
    PluginProcessor.h
    PluginProcessor.cpp
    PluginEditor.cpp   
    PluginEditor.h  
)

# ==========================================
# EXPLICIT LINKING
# ==========================================
# Link only what is necessary directly to the plugin target, optimizing compile time
target_link_libraries(IbanezTubeScreamer PRIVATE
    juce::juce_audio_basics
    juce::juce_audio_processors
    juce::juce_audio_utils
    juce::juce_data_structures
    juce::juce_dsp
    juce::juce_gui_basics
)

target_compile_definitions(IbanezTubeScreamer PUBLIC
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
)

# ==========================================
# LINUX SPECIFIC CONFIGURATION (Target-Based)
# ==========================================
if(UNIX AND NOT APPLE)
    find_package(PkgConfig REQUIRED)
    
    # GTK3 is usually required for the windowing/graphics backend (juce_gui_basics) on Linux
    pkg_check_modules(GTK3 REQUIRED gtk+-3.0)
    target_include_directories(IbanezTubeScreamer PRIVATE ${GTK3_INCLUDE_DIRS})
    target_link_libraries(IbanezTubeScreamer PRIVATE ${GTK3_LIBRARIES})

    # WebKit and Soup REMOVED from the main flow since JUCE_WEB_BROWSER is disabled (0).
    # If you need them in the future, link them explicitly here:
    # pkg_check_modules(WEBKIT REQUIRED webkit2gtk-4.1)
    # target_link_libraries(IbanezTubeScreamer PRIVATE ${WEBKIT_LIBRARIES})
endif()
